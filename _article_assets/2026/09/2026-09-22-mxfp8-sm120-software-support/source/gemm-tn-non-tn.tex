% Shape/geometry ledger: M=2, K=3, N=4 illustrative cells.
% Each occurrence of an axis uses 0.40 cm per cell, including transposes.

% Tile lightness is illustrative texture, not a measured numeric value.
% Each row is a separate layout example. Arrows map input -> op(input).
\newcommand{\CellPitch}{0.40}
\newcommand{\TileGap}{0.045}

% Matrix face: center x/y, rows, columns, role color, transpose-texture flag,
% texture family (1: effective A, 2: effective B, 3: C).
\newcommand{\MatrixFace}[7]{%
  \begingroup
  \pgfmathsetmacro{\facewidth}{#4*\CellPitch}
  \pgfmathsetmacro{\faceheight}{#3*\CellPitch}
  \pgfmathsetmacro{\leftx}{#1-\facewidth/2}
  \pgfmathsetmacro{\topy}{#2+\faceheight/2}
  \pgfmathtruncatemacro{\lastrow}{#3-1}
  \pgfmathtruncatemacro{\lastcol}{#4-1}
  \foreach \r in {0,...,\lastrow}{%
    \foreach \c in {0,...,\lastcol}{%
      \pgfmathtruncatemacro{\rr}{#6==1 ? \c : \r}
      \pgfmathtruncatemacro{\cc}{#6==1 ? \r : \c}
      \ifnum#7=1
        \pgfmathtruncatemacro{\entryindex}{\rr*3+\cc}
        \pgfmathtruncatemacro{\tone}{{35,80,55,80,35,55}[\entryindex]}
      \else\ifnum#7=2
        \pgfmathtruncatemacro{\entryindex}{\rr*4+\cc}
        \pgfmathtruncatemacro{\tone}{{55,35,80,55,35,80,55,80,80,55,35,55}[\entryindex]}
      \else
        \pgfmathtruncatemacro{\entryindex}{\rr*4+\cc}
        \pgfmathtruncatemacro{\tone}{{35,55,35,80,80,35,80,80}[\entryindex]}
      \fi\fi
      \fill[#5!\tone,rounded corners=.65pt]
        ({\leftx+\c*\CellPitch+\TileGap/2},
         {\topy-(\r+1)*\CellPitch+\TileGap/2})
        rectangle ++({\CellPitch-\TileGap},{\CellPitch-\TileGap});
    }%
  }%
  % Neutral brackets surround the face without changing its axis geometry.
  \draw[black!62,line width=.65pt]
    ({\leftx+.07},{\topy+.025}) -- ({\leftx-.045},{\topy+.025})
    -- ({\leftx-.045},{\topy-\faceheight-.025})
    -- ({\leftx+.07},{\topy-\faceheight-.025});
  \draw[black!62,line width=.65pt]
    ({\leftx+\facewidth-.07},{\topy+.025})
    -- ({\leftx+\facewidth+.045},{\topy+.025})
    -- ({\leftx+\facewidth+.045},{\topy-\faceheight-.025})
    -- ({\leftx+\facewidth-.07},{\topy-\faceheight-.025});
  \endgroup
}

% A full comparison row: y, label, A transpose flag, B transpose flag.
\newcommand{\GemmRow}[4]{%
  \node[font=\small\bfseries,text=black!65] at (.20,#1) {#2};
  \ifnum#3=1
    \MatrixFace{1.85}{#1}{3}{2}{tensorA}{1}{1}
    \def\ashape{K\times M}\def\aop{A^T}\def\aflag{T}
  \else
    \MatrixFace{1.85}{#1}{2}{3}{tensorA}{0}{1}
    \def\ashape{M\times K}\def\aop{A}\def\aflag{N}
  \fi
  \ifnum#4=1
    \MatrixFace{4.00}{#1}{4}{3}{tensorB}{1}{2}
    \def\bshape{N\times K}\def\bop{B^T}\def\bflag{T}
  \else
    \MatrixFace{4.00}{#1}{3}{4}{tensorB}{0}{2}
    \def\bshape{K\times N}\def\bop{B}\def\bflag{N}
  \fi
  \node[symbol] at (1.85,{#1-1.15}) {$A$};
  \node[dimlabel] at (1.85,{#1-1.62}) {$\ashape$};
  \node[symbol] at (4.00,{#1-1.15}) {$B$};
  \node[dimlabel] at (4.00,{#1-1.62}) {$\bshape$};
  \begin{pgfonlayer}{background}
    \draw[flow] (5.22,#1) -- (6.52,#1);
  \end{pgfonlayer}
  \node[font=\scriptsize,text=black!65,fill=white,inner sep=2pt]
    at (5.87,{#1+.40}) {$\aflag\,,\ \bflag$};
  \MatrixFace{7.80}{#1}{2}{3}{tensorA}{0}{1}
  \MatrixFace{10.40}{#1}{3}{4}{tensorB}{0}{2}
  \MatrixFace{13.35}{#1}{2}{4}{tensorC}{0}{3}
  \node[operator] at (9.10,#1) {$\times$};
  \node[operator] at (11.95,#1) {$=$};
  \node[symbol] at (7.80,{#1-1.15}) {$\aop$};
  \node[dimlabel] at (7.80,{#1-1.62}) {$M\times K$};
  \node[symbol] at (10.40,{#1-1.15}) {$\bop$};
  \node[dimlabel] at (10.40,{#1-1.62}) {$K\times N$};
  \node[symbol] at (13.35,{#1-1.15}) {$C$};
  \node[dimlabel] at (13.35,{#1-1.62}) {$M\times N$};
}
